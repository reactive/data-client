```ts title="TodoResource" collapsed
import { Entity, resource } from '@data-client/rest';

export class Todo extends Entity {
  id = 0;
  title = '';
  completed = false;

  static key = 'Todo';
}
export const TodoResource = resource({
  path: '/todos/:id',
  schema: Todo,
  optimistic: true,
});
```

```ts title="ToastManager" framework-imports
import {
  type Manager,
  type Middleware,
  actionTypes,
} from '@data-client/react';

// toast is the app's toast library, e.g. import { toast } from 'sonner'
export default class ToastManager implements Manager {
  middleware: Middleware = controller => next => async action => {
    if (
      action.type === actionTypes.SET_RESPONSE &&
      action.endpoint.sideEffect
    ) {
      if (action.error) toast.error(`${action.endpoint.name} failed`);
      else toast.success(`${action.endpoint.name} succeeded`);
    }
    return next(action);
  };

  cleanup() {}
}
```

```tsx title="TodoList"
import { useController, useSuspense } from '@data-client/react';
import { TodoResource } from './TodoResource';

function TodoList() {
  const ctrl = useController();
  const todos = useSuspense(TodoResource.getList);
  return (
    <div className="list">
      {todos.map(todo => (
        <div key={todo.pk()} className="listItem nogap">
          <label>
            <input
              type="checkbox"
              checked={todo.completed}
              onChange={e =>
                ctrl
                  .fetch(
                    TodoResource.partialUpdate,
                    { id: todo.id },
                    { completed: e.currentTarget.checked },
                  )
                  // ToastManager reports failures
                  .catch(() => {})
              }
            />
            {todo.title}
          </label>
        </div>
      ))}
    </div>
  );
}
render(
  <>
    <header className="appHeader">
      <h1 className="appTitle">Launch tasks</h1>
    </header>
    <TodoList />
    <Toaster />
  </>,
);
```

```html title="TodoList.vue"
<script setup lang="ts">
  import { useController, useSuspense } from '@data-client/vue';
  import { TodoResource } from './TodoResource';

  const ctrl = useController();
  const todos = await useSuspense(TodoResource.getList);

  const toggle = (id: number, e: Event) =>
    ctrl
      .fetch(
        TodoResource.partialUpdate,
        { id },
        { completed: (e.currentTarget as HTMLInputElement).checked },
      )
      // ToastManager reports failures
      .catch(() => {});
</script>

<template>
  <header class="appHeader">
    <h1 class="appTitle">Launch tasks</h1>
  </header>
  <div class="list">
    <div v-for="todo in todos" :key="todo.pk()" class="listItem nogap">
      <label>
        <input
          type="checkbox"
          :checked="todo.completed"
          @change="toggle(todo.id, $event)"
        />
        {{ todo.title }}
      </label>
    </div>
  </div>
  <Toaster />
</template>
```
