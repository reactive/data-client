  completed = false;

}

```tsx title="TodoItem" collapsed
import { useController } from '@data-client/react';
import { TodoResource, type Todo } from './TodoResource';

export default function TodoItem({ todo }: { todo: Todo }) {
  const ctrl = useController();
  const handleChange = e =>
    ctrl.fetch(
      TodoResource.partialUpdate,
      { id: todo.id },
      { completed: e.currentTarget.checked },
    );
  return (
    <div className="listItem nogap">
      <label>
        <input
          type="checkbox"
          checked={todo.completed}
          onChange={handleChange}
        />
        {todo.completed ? <strike>{todo.title}</strike> : todo.title}
      </label>
    </div>
  );
}
```

```tsx title="Session" collapsed
import { useSuspense } from '@data-client/react';
import { TodoResource } from './TodoResource';
import TodoItem from './TodoItem';

export default function Session() {
  const todos = useSuspense(TodoResource.getList, { userId: '1' });
  const [note, setNote] = React.useState('');
  return (
    <div>
      <TextInput
        size="small"
        placeholder="local-only note (lost on crash)"
        value={note}
        onChange={e => setNote(e.currentTarget.value)}
      />
      {todos.map(todo => (
        <TodoItem key={todo.pk()} todo={todo} />
      ))}
    </div>
  );
}
```

import { useController } from '@data-client/react';

function App() {
  const ctrl = useController();
  const [session, setSession] = React.useState(0);
  const handleCrash = async () => {
    await ctrl.resetEntireStore();
    setSession(s => s + 1);
  };
  return (
    <div>
      <button onClick={handleCrash}>Simulate crash</button>
      <AsyncBoundary fallback={<Loading />}>
        <Session key={session} />
      </AsyncBoundary>
    </div>
  );
}
