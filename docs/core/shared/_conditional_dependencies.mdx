import CodeBlock from '@theme/CodeBlock';

::::tip[Conditional Dependencies]

Use `null` as the second argument to any <abbr title="Reactive Data Client">Data Client</abbr> hook means "do nothing."

:::react

<CodeBlock
  language="typescript">
  {`// todo could be undefined if id is undefined
const todo = ${props.hook ?? 'useSuspense'}(TodoResource.get, id ? { id } : null);`}
</CodeBlock>

:::

:::vue

<CodeBlock
  language="typescript">
  {`// todo could be undefined if id is undefined
const todo = ${(props.hook ?? 'useSuspense') === 'useSuspense' ? 'await ' : ''}${props.hook ?? 'useSuspense'}(
  TodoResource.get,
  computed(() => (id.value ? { id: id.value } : null)),
);`}
</CodeBlock>

:::
