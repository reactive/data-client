```ts title="Resources" collapsed
import { Entity, resource } from '@data-client/rest';

export class Issue extends Entity {
  id = 0;
  assigneeId = 0;
  title = '';
  body = '';
  state: 'open' | 'closed' = 'open';

  static key = 'Issue';
}
export const IssueResource = resource({
  path: '/issues/:id',
  searchParams: {} as { assigneeId?: string | number } | undefined,
  schema: Issue,
  optimistic: true,
});
```

```tsx title="IssueDetail" route="/issues/:id" {5} collapsed
import { useSuspense } from '@data-client/react';
import { IssueResource } from './Resources';

export default function IssueDetail({ id }: { id: string }) {
  const issue = useSuspense(IssueResource.get, { id });
  return (
    <div>
      <h2>{issue.title}</h2>
      <p className="meta">{issue.state === 'open' ? 'Open' : 'Closed'}</p>
      <p>{issue.body}</p>
    </div>
  );
}
```

```tsx title="IssueList" route="/" {5}
import { useSuspense } from '@data-client/react';
import { IssueResource } from './Resources';

export default function IssueList() {
  const issues = useSuspense(IssueResource.getList);
  return (
    <div className="list">
      {issues.map(issue => (
        <Link
          key={issue.pk()}
          className="listItem"
          to={`/issues/${issue.id}`}
        >
          {issue.title}
          <span className="meta">
            {issue.state === 'open' ? 'Open' : 'Closed'}
          </span>
        </Link>
      ))}
    </div>
  );
}
```
