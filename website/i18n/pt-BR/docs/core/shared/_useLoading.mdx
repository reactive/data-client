```tsx title="PostForm" collapsed
export default function PostForm({ onSubmit, loading, error }) {
  const handleSubmit = e => {
    e.preventDefault();
    const data = new FormData(e.target);
    onSubmit(data);
  };
  return (
    <form onSubmit={handleSubmit} aria-busy={loading}>
      <fieldset disabled={loading}>
        <label className="field">
          Title
          <input type="text" name="title" defaultValue="My New Post" required />
        </label>
        <label className="field">
          Body
          <textarea
            name="body"
            rows={12}
            required
            defaultValue="After clicking 'save', the form is locked and the button animates until the POST completes. Upon completion the newly created post is displayed immediately as Reactive Data Client is able to use the fetch response to populate the store."
          />
        </label>
        {error ? (
          <div className="alert alert--danger">{error.message}</div>
        ) : null}
        <button type="submit">{loading ? 'Saving…' : 'Save'}</button>
      </fieldset>
    </form>
  );
}
```

```tsx title="PostCreate" route="/" {8}
import { useLoading, useController } from '@data-client/react';
import { PostResource } from './PostResource';
import PostForm from './PostForm';

export default function PostCreate() {
  const ctrl = useController();
  const navigate = useNavigate();
  const [handleSubmit, loading, error] = useLoading(
    async data => {
      const post = await ctrl.fetch(PostResource.getList.push, data);
      navigate(`/posts/${post.id}`);
    },
    [ctrl, navigate],
  );
  return (
    <PostForm onSubmit={handleSubmit} loading={loading} error={error} />
  );
}
```
